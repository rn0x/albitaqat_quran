البدء السريع

ابدأ باستخدام البيانات في دقائق

# تحميل البيانات عبر Python
pip install -r src/requirements.txt

# تحميل جميع الملفات
python src/downloader.py --mode full

# تحميل الصوت فقط
python src/downloader.py --mode audio

# تحميل PDF فقط
python src/downloader.py --mode pdf

# التحقق من الملفات
python src/downloader.py --mode verify
# تشغيل API محلياً
cd api
npm install
npm run dev

# الاختبار
npm test

# بناء للإنتاج
npm start
# بناء وتشغيل Docker
docker build -t albitaqat-api -f api/Dockerfile .
docker run -p 3000:3000 albitaqat-api

# أو عبر docker-compose
cd api
docker compose up -d --build

# عرض السجلات
docker compose logs -f
# استدعاء API
curl http://localhost:3000/api/surahs

# سورة محددة
curl http://localhost:3000/api/surahs/1

# البحث
curl "http://localhost:3000/api/surahs/search?q=بقرة"

# إحصائيات
curl http://localhost:3000/api/stats
