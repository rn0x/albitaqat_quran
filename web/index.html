Python

import json

# قراءة بيانات السور
with open('data/quran_cards_full.json') as f:
    data = json.load(f)

# عرض جميع السور
for surah in data['surahs']:
    print(f"{surah['number']} - {surah['name_arabic']}")

# البحث عن سورة
baqarah = next(
    s for s in data['surahs']
    if s['number'] == 2
)
print(baqarah['downloads']['audio']['url'])

Node.js

const fs = require('fs');

// قراءة بيانات السور
const data = JSON.parse(
  fs.readFileSync('data/quran_cards_full.json')
);

// عرض جميع السور
data.surahs.forEach(surah => {
  console.log(`${surah.number} - ${surah.name_arabic}`);
});

// جلب سورة عبر API
const res = await fetch('http://localhost:3000/api/surahs/1');
const surah = await res.json();
console.log(surah.data.name_english);

JavaScript (Browser)

// جلب البيانات من API
const response = await fetch(
  'http://localhost:3000/api/surahs'
);
const { data } = await response.json();

// عرض السور في الصفحة
const list = document.getElementById('surahs');
data.forEach(surah => {
  list.innerHTML += `
    <div class="surah-card">
      <h3>${surah.name_arabic}</h3>
      <p>${surah.name_english}</p>
      <a href="${surah.downloads.audio.url}">
        استمع
      </a>
    </div>
  `;
});
